  <script>
    const numberEl = document.getElementById('number');
    const pulse = document.getElementById('pulse');
    let current = 1;

    function show(n){
      // scale pop
      numberEl.style.transform = 'scale(.92)';
      setTimeout(()=> numberEl.style.transform = 'scale(1)', 120);

      // flash pulse quickly
      pulse.style.transform = 'scale(1)';
      pulse.style.opacity = '1';
      setTimeout(()=>{ pulse.style.transform = 'scale(1.18)'; pulse.style.opacity='0.06'; }, 80);

      numberEl.textContent = n;
    }

    window.addEventListener('keydown', (e)=>{
      if(e.key === 'Enter'){
        current = current === 4 ? 1 : current + 1;
        show(current);
      } else if(e.key === 'Backspace' || e.key === 'ArrowLeft'){
        e.preventDefault();
        current = current === 1 ? 4 : current - 1;
        show(current);
      } else if(e.key === 'ArrowRight'){
        current = current === 4 ? 1 : current + 1;
        show(current);
      }
    });

    // initial subtle pulse
    setTimeout(()=> show(current), 300);
  </script>
